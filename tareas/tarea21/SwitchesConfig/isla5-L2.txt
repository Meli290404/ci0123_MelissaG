! ============================================================
!  l2 2960 - isla 5 - grupo 2
!  red de la isla: 172.16.123.80/28, vlan 250
!  .81 = l3 (gateway)   .82 = este switch   .95 = broadcast
!  el gigabitethernet0/1 de este switch se conecta al fastethernet0/5 del l3
! ============================================================

enable
configure terminal
hostname SW-I5-G2
no ip domain-lookup

! los mensajes del switch no interrumpen lo que se esta escribiendo en la consola
line console 0
 logging synchronous
exit

! ---------- red de la isla ----------
vlan 250
 name VLAN250-ISLA5
exit

! interfaz virtual de la vlan: solo sirve para llegar al switch (ping, ssh), no rutea
interface vlan 250
 description administracion del switch de la isla 5
 ip address 172.16.123.82 255.255.255.240
 no shutdown
exit

! hacia donde manda el switch lo que va para otra red
ip default-gateway 172.16.123.81

! ---------- enlace con el l3 ----------
! troncal 802.1q; aunque solo lleva la vlan 250, la etiqueta permite que el l3 sepa de que isla viene
interface GigabitEthernet0/1
 description enlace al l3 del grupo 2 (fa0/5)
 switchport mode trunk
 switchport trunk allowed vlan 250
exit

! ---------- puertos de las computadoras ----------
! cada puerto pertenece solo a la vlan 250; portfast porque ahi no se conectan otros switches
interface range FastEthernet0/1 - 12
 description computadoras de la isla 5
 switchport mode access
 switchport access vlan 250
 spanning-tree portfast
exit

! ---------- dhcp ----------
! primero se apartan las ips fijas (l3 y switch) para que el dhcp no las entregue
ip dhcp excluded-address 172.16.123.81 172.16.123.82
ip dhcp pool POOL-ISLA5
 network 172.16.123.80 255.255.255.240
 default-router 172.16.123.81
exit

end
write memory

! ---------- comprobar ----------
! show interfaces status          fa0/1-12 en la vlan 250, gi0/1 como trunk
! show interfaces trunk           vlan 250 permitida en gi0/1
! show ip interface brief         vlan250 up/up con la .82
! ping 172.16.123.81           llega al l3
! show ip dhcp binding            computadoras que ya recibieron ip
